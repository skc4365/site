## 완성 심플 예제

허용된 Tool만 실행하는 작은 Agent.

```python
from collections.abc import Callable


def get_temperature(city: str) -> str:
    return f"{city}: 23°C"


def search_manual(keyword: str) -> str:
    return f"매뉴얼 검색: {keyword}"


TOOLS: dict[str, Callable[[str], str]] = {
    "weather": get_temperature,
    "manual": search_manual,
}


def choose_tool(question: str) -> tuple[str, str]:
    # 판단 단계
    if "날씨" in question:
        return "weather", question.replace("날씨", "").strip()
    return "manual", question


def run_agent(question: str) -> str:
    tool_name, argument = choose_tool(question)
    tool = TOOLS[tool_name]
    return tool(argument)


print(run_agent("서울 날씨"))
print(run_agent("프레스 점검"))
```
