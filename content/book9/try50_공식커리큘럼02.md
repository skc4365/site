## 완성 심플 예제

외부 API 없이 Harness 구조 확인.

```python
from pydantic import BaseModel, Field, ValidationError


class Summary(BaseModel):
    title: str = Field(min_length=1, max_length=30)
    keywords: list[str] = Field(min_length=1, max_length=3)


def build_prompt(text: str) -> str:
    return f"""
역할: 문서 요약기
규칙: 제목과 핵심어 3개 이하
문서: <DATA>{text}</DATA>
출력: title, keywords를 가진 JSON
""".strip()


def fake_llm(prompt: str) -> dict:
    # 모델 응답 예시
    return {"title": "설비 점검", "keywords": ["센서", "예방", "정비"]}


def run(text: str) -> Summary:
    prompt = build_prompt(text)
    raw_output = fake_llm(prompt)
    return Summary.model_validate(raw_output)


try:
    print(run("센서 데이터로 설비 고장을 예방한다."))
except ValidationError as error:
    print("출력 검증 실패", error)
```
